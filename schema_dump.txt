=== TABLE: investors ===
- id (int(10) unsigned)
- full_name (varchar(150))
- phone (varchar(30))
- city (varchar(100))
- address (varchar(255))
- notes (text)
- national_id (varchar(30))
- contract_path (varchar(255))
- id_card_path (varchar(255))
- created_at (datetime)
- updated_at (datetime)

=== TABLE: users ===
- id (int(10) unsigned)
- investor_id (int(10) unsigned)
- role (enum('admin','staff','investor'))
- username (varchar(100))
- password_hash (varchar(255))
- last_login_at (datetime)
- created_at (datetime)
- updated_at (datetime)

=== TABLE: withdraw_requests ===
- id (int(10) unsigned)
- investor_id (int(10) unsigned)
- amount (decimal(12,2))
- currency (varchar(10))
- request_date (datetime)
- status (enum('pending','approved','rejected'))
- note (text)
- created_at (datetime)
- updated_at (datetime)

=== TABLE: deposit_types ===
- id (int(10) unsigned)
- name_ar (varchar(100))
- code (varchar(50))
- profit_rate_monthly (decimal(8,5))
- min_days (int(11))
- max_days (int(11))
- created_at (datetime)
- updated_at (datetime)
- min_rate (decimal(8,5))
- max_rate (decimal(8,5))

=== TABLE: deposits ===
- id (int(10) unsigned)
- investor_id (int(10) unsigned)
- deposit_type_id (int(10) unsigned)
- amount (decimal(12,2))
- start_date (date)
- end_date (date)
- profit_rate_monthly (decimal(8,5))
- last_profit_date (date)
- status (enum('active','completed','cancelled','defaulted'))
- created_at (datetime)
- updated_at (datetime)
- accumulated_profit (decimal(12,2))
- last_withdrawal_date (date)

=== TABLE: transactions ===
- id (int(10) unsigned)
- receipt_no (varchar(30))
- investor_id (int(10) unsigned)
- deposit_id (int(10) unsigned)
- type (enum('deposit','profit','withdraw'))
- amount (decimal(12,2))
- date (datetime)
- note (varchar(255))
- created_at (datetime)

=== TABLE: monthly_rates ===
- id (int(10) unsigned)
- month (varchar(7))
- deposit_type_id (int(10) unsigned)
- rate_percent (decimal(8,5))
- created_at (datetime)

=== TABLE: profit_cycles ===
- id (int(10) unsigned)
- deposit_id (int(10) unsigned)
- cycle_date (date)
- processed_at (datetime)
